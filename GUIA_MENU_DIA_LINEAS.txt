     8	    {
     9	        static $cache = null;
    10	        if ($cache !== null) {
    11	            return $cache;
    12	        }
    13	
    14	        global $conn;
    15	        $result = $conn->query("SHOW COLUMNS FROM ofertas LIKE 'es_menu_dia'");
    16	        $cache = $result && $result->num_rows > 0;
    17	        if ($result) {
    18	            $result->free();
    19	        }
    20	
    21	        if (!$cache) {
    22	            // Compatibilidad automática en instalaciones antiguas
    23	            $conn->query("ALTER TABLE ofertas ADD COLUMN es_menu_dia TINYINT(1) NOT NULL DEFAULT 0");
    24	            $retry = $conn->query("SHOW COLUMNS FROM ofertas LIKE 'es_menu_dia'");
    25	            $cache = $retry && $retry->num_rows > 0;
    26	            if ($retry) {
    27	                $retry->free();
    28	            }
    29	        }
    30

   148	2) RESUMEN DEL FLUJO FINAL
   149	==========================
   150	1. Gerente crea/edita oferta.
   151	2. Marca checkbox "Es menú del día".
   152	3. Se persiste `es_menu_dia` en tabla ofertas.
   153	4. Cliente entra a catálogo y pulsa "Ver menú del día".
   154	5. Ve detalle completo en `menuDia.php`.
   155	6. Pulsa "Quiero el menú del día".
   156	7. Se añaden productos al carrito y se aplica la oferta.
   157	8. En carrito se refleja el precio final del menú (vía descuento de oferta).
   158	
   159	
   160	3) NOTAS DE COMPATIBILIDAD
   161	==========================
   162	- Si la BD es antigua y no tiene `es_menu_dia`, el DAO intenta crearla automáticamente.
   163	- Si no hay menú del día activo, la vista `menuDia.php` informa que no hay menú disponible.
   164	- El menú del día se identifica por flag explícito (`es_menu_dia`), no por nombre.
